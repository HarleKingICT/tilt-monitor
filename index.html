const THRESHOLD = 5; // 傾斜閾值（度），可根據需要調整 let isMonitoring = false; let isAlarming = false; // 新增旗標：追蹤警報是否已啟動 let alarmAudio = document.getElementById('alarm-sound'); let statusDiv = document.getElementById('status'); let tiltInfoDiv = document.getElementById('tilt-info'); let alertDiv = document.getElementById('alert'); // 請求權限（針對iOS） async function requestPermission() { if (typeof DeviceOrientationEvent.requestPermission === 'function') { try { const permission = await DeviceOrientationEvent.requestPermission(); console.log('Permission:', permission); // Debug: 檢查權限結果 if (permission === 'granted') { return true; } else { alert('權限被拒絕，無法使用陀螺儀。'); return false; } } catch (error) { console.error('請求權限失敗:', error); return false; } } return true; // Android 等不需要請求 } // 計算傾斜度（使用 beta 和 gamma 的組合） function calculateTilt(beta, gamma) { // 計算總傾斜角度（歐幾里德範數） return Math.sqrt(beta * beta + gamma * gamma); } // 處理方向事件 function handleOrientation(event) { if (!isMonitoring) return; console.log('Event fired:', event.beta, event.gamma); // Debug: 檢查事件是否觸發及值 const beta = event.beta || 0; // 前後傾斜（度） const gamma = event.gamma || 0; // 左右傾斜（度） const tilt = calculateTilt(beta, gamma); tiltInfoDiv.textContent = `傾斜度：${tilt.toFixed(2)}°`; if (tilt > THRESHOLD) { alertDiv.style.display = 'block'; document.body.style.backgroundColor = '#ffcccc'; if (!isAlarming) { // 只在第一次進入警報時播放 alarmAudio.loop = true; alarmAudio.play(); isAlarming = true; } if ('vibrate' in navigator) { navigator.vibrate([200, 100, 200]); // 振動模式，可以視需要調整為不重複 } } else { alertDiv.style.display = 'none'; document.body.style.backgroundColor = '#f0f0f0'; if (isAlarming) { // 只在離開警報時停止 alarmAudio.pause(); alarmAudio.currentTime = 0; alarmAudio.loop = false; // 可選，重置循環 isAlarming = false; } } } // 啟動按鈕事件 document.getElementById('start-btn').addEventListener('click', async () => { if (window.DeviceOrientationEvent) { console.log('DeviceOrientation supported'); // Debug: 檢查是否支援 } else { alert('您的設備不支援陀螺儀'); return; } const hasPermission = await requestPermission(); if (hasPermission) { window.addEventListener('deviceorientation', handleOrientation); isMonitoring = true; statusDiv.textContent = '狀態：監測中'; } });
